final st = await c.read(settingsProvider.future);
final entries = await c.read(allJournalEntriesProvider.future);
final dreams = await c.read(allDreamEntriesProvider.future);
final h = st.dayStartHour;
String k(DateTime d) => d.month.toString() + '/' + d.day.toString();
final streakVals = await c.read(trackerValuesProvider(kStreakTrackerId).future);
final streakStr = streakVals.where((v) => v.periodStart.isAfter(DateTime(2026, 10, 11))).map((v) => k(v.periodStart) + '=' + v.intValue!.toInt().toString()).join(' ');
final jeVals = await c.read(trackerValuesProvider(kJournalEntriesTrackerId).future);
final jeStr = (jeVals.map((v) => v.periodStart).where((d) => d.year == 2026).toList()..sort()).map(k).join(',');
final dlVals = await c.read(trackerValuesProvider(kDreamLoggedTrackerId).future);
final dlStr = (dlVals.map((v) => v.periodStart).toList()..sort()).map(k).join(',');
final wcVals = await c.read(trackerValuesProvider(kWordCountTrackerId).future);
final wcStr = wcVals.where((v) => v.periodStart.isAfter(DateTime(2026, 10, 11)) && v.intValue! > 0).map((v) => k(v.periodStart) + '=' + v.intValue!.toInt().toString()).join(' ');
final best = c.read(periodicPromptServiceProvider).longestJournalStreak(entries, dayStartHour: h);
final now = DateTime.now();
debugPrint('DSPROBE hour=' + h.toString() + ' now=' + now.toString() + ' best=' + best.toString());
debugPrint('DSPROBE streak ' + streakStr);
debugPrint('DSPROBE journaled2026 ' + jeStr);
debugPrint('DSPROBE dreamLogged ' + dlStr);
debugPrint('DSPROBE words ' + wcStr);
debugPrint('DSPROBE dreams=' + dreams.length.toString() + ' entries=' + entries.length.toString());
