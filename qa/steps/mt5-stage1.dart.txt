String s(GaiaRecallCandidate x) => x.strength.toStringAsFixed(2) + ' ' + (x.memory.text.length > 50 ? x.memory.text.substring(0, 50) : x.memory.text);
for (final text in ['when is Biscuit due for shots', 'I have been sleeping badly', 'I\'ve been sleeping badly', 'my sleep has been bad lately']) {
